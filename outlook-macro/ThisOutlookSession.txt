Private WithEvents mInspectors As Outlook.Inspectors
Private mWatchers As New Collection

Private Sub Application_Startup()
    Set mInspectors = Application.Inspectors
End Sub

Private Sub mInspectors_NewInspector(ByVal Inspector As Outlook.Inspector)
    If TypeName(Inspector.CurrentItem) <> "MailItem" Then Exit Sub
    Dim w As clsDraftWatch
    Set w = New clsDraftWatch
    Set w.Insp = Inspector
    mWatchers.Add w
End Sub
