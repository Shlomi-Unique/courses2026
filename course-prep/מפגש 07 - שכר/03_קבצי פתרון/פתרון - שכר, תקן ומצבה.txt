שלב 3: מדדים (מפגש 5)
-----------------------------------------------------------
עלות מעביד := SUM ( tblPayroll[עלות מעביד] )

תקציב שכר := SUM ( tblHeadcountPlan[תקציב עלות מעביד] )

סטייה לטובה שכר := [תקציב שכר] - [עלות מעביד]

התאמה ליומן :=
[עלות מעביד] - CALCULATE ( [ביצוע], tblAccount[כרטיס חשבון] = 6100 )
  בדיקה: 0 בכל חודש ובכל שנה (בלי כלי פריסה של מחלקה).
  עם כלי פריסה של מחלקה ההתאמה "נשברת", כי ביומן כל השכר רשום על הנהלה. זה בדיוק המסר.

מצבה :=
VAR MonthStart = MIN ( DimDate[תאריך] )
VAR MonthEnd = MAX ( DimDate[תאריך] )
RETURN
    SUMX (
        FILTER (
            tblEmployees,
            tblEmployees[תאריך תחילה] <= MonthEnd
                && ( ISBLANK ( tblEmployees[תאריך עזיבה] ) || tblEmployees[תאריך עזיבה] >= MonthStart )
        ),
        tblEmployees[היקף משרה]
    )

תקן := SUM ( tblHeadcountPlan[תקן] )

מצבה ממוצעת :=
AVERAGEX ( FILTER ( VALUES ( DimDate[חודש] ), [עלות מעביד] > 0 ), [מצבה] )

תקן ממוצע :=
AVERAGEX ( FILTER ( VALUES ( DimDate[חודש] ), [עלות מעביד] > 0 ), [תקן] )

עלות חודשית למשרה :=
DIVIDE ( [עלות מעביד], [מצבה ממוצעת] * COUNTROWS ( FILTER ( VALUES ( DimDate[חודש] ), [עלות מעביד] > 0 ) ) )

עובדים ששולמו := DISTINCTCOUNT ( tblPayroll[מס עובד] )
  (המלכודת: מכירות 08/2025 = 2 עובדים ששולמו, אבל מצבה = 1. נועה עזבה ב-31/07 ושולמה גם באוגוסט.)

הערה: "מצבה ממוצעת" נספרת רק בחודשים שיש בהם תלושים (עד 08/2026).

שלב 5: חריגות בנוסחאות מערך (מפגש 6), גיליון "בקרות"
-----------------------------------------------------------
1. שולם אחרי עזיבה:
=LET(
    leave, XLOOKUP(tblPayroll[מס עובד], tblEmployees[מס עובד], tblEmployees[תאריך עזיבה]),
    FILTER(tblPayroll, (N(leave) > 0) * (tblPayroll[חודש] > leave), "אין")
)
   בדיקה: 1 שורה. עובד 103, 08/2025, עלות מעביד 9,007.50.
   (N() הופך תא ריק ל-0, כדי שעובדים שלא עזבו לא ייכנסו.)

2. תשלום כפול:
=FILTER(tblPayroll, COUNTIFS(tblPayroll[מס עובד], tblPayroll[מס עובד], tblPayroll[חודש], tblPayroll[חודש]) > 1, "אין")
   בדיקה: 2 שורות זהות. עובד 104, 11/2025, 8,647.20 כל אחת.

3. פנסיה חסרה:
=FILTER(tblPayroll, tblPayroll[פנסיה מעביד] = 0, "אין")
   בדיקה: 3 שורות. עובדת 105, 03 עד 05/2026. חסר 321.36 בכל חודש, 964.08 בסך הכל.

סיכום: שולם ביתר 17,654.70 (עזיבה + כפול). לא הופרש לפנסיה 964.08.
